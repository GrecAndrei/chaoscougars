===FILE: app.js===
(function () {
  'use strict';

  // ============== DOM REFERENCES ==============
  const timerBar      = document.getElementById('timer-bar');
  const timerFill     = document.getElementById('timer-fill');
  const effectsList   = document.getElementById('effects-list');
  const votePanel     = document.getElementById('vote-panel');
  const voteTime      = document.getElementById('vote-time');
  const voteOptions   = document.getElementById('vote-options');
  const hitContainer  = document.getElementById('hit-container');
  const cougarCounter = document.getElementById('cougar-counter');
  const cougarCount   = document.getElementById('cougar-count');
  const phaseOverlay  = document.getElementById('phase-overlay');
  const phaseText     = document.getElementById('phase-text');

  // ============== STATE ==============
  const effectsState = new Map();
  const voteState = {
    visible: false,
    elements: [],     // [{ nameEl, fillEl }]
    pendingHide: null
  };
  let lastCougarCount = 0;
  let currentPhase = null;

  // ============== HELPERS ==============
  function clamp(v, min, max) {
    return Math.max(min, Math.min(max, v));
  }

  function formatSeconds(s) {
    const v = Math.max(0, Math.ceil(Number(s) || 0));
    return v + 's';
  }

  // ============== TIMER ==============
  function updateTimer(progress) {
    const p = clamp(Number(progress) || 0, 0, 1);
    timerFill.style.width = (p * 100) + '%';
    if (p < 0.2) {
      timerBar.classList.add('urgent');
    } else {
      timerBar.classList.remove('urgent');
    }
  }

  // ============== EFFECTS LIST ==============
  function updateEffects(list) {
    if (!Array.isArray(list)) list = [];
    const maxEffects = 3;
    const display = list.slice(0, maxEffects);
    const newNames = new Set();

    display.forEach(function (eff) {
      if (eff && eff.name) newNames.add(eff.name);
    });

    // Mark missing ones for removal
    effectsState.forEach(function (data, name) {
      if (!newNames.has(name)) {
        if (data.element && data.element.parentNode && !data.removing) {
          data.removing = true;
          data.element.classList.add('removing');
          const el = data.element;
          setTimeout(function () {
            if (el.parentNode) el.parentNode.removeChild(el);
          }, 320);
        }
        effectsState.delete(name);
      }
    });

    // Add or update
    display.forEach(function (eff) {
      if (!eff || !eff.name) return;
      const remaining = Math.max(0, Number(eff.remaining) || 0);
      const existing = effectsState.get(eff.name);
      if (existing && !existing.removing) {
        existing.timeEl.textContent = formatSeconds(remaining);
      } else {
        const card = document.createElement('div');
        card.className = 'effect-card';

        const nameEl = document.createElement('span');
        nameEl.className = 'effect-name';
        nameEl.textContent = eff.name;

        const timeEl = document.createElement('span');
        timeEl.className = 'effect-time';
        timeEl.textContent = formatSeconds(remaining);

        card.appendChild(nameEl);
        card.appendChild(timeEl);
        effectsList.appendChild(card);

        effectsState.set(eff.name, {
          element: card,
          timeEl: timeEl,
          removing: false
        });
      }
    });

    // Cap visible cards in DOM (defensive: extra cleanup if list grew)
    while (effectsList.children.length > maxEffects) {
      const last = effectsList.lastChild;
      if (last) effectsList.removeChild(last);
    }
  }

  // ============== VOTE ==============
  function buildVoteOptions(options) {
    voteOptions.innerHTML = '';
    voteState.elements = [];
    const max = Math.min(3, options.length);
    for (let i = 0; i < max; i++) {
      const opt = options[i];
      if (!opt || !opt.name) continue;

      const optEl = document.createElement('div');
      optEl.className = 'vote-option';

      const keyEl = document.createElement('div');
      keyEl.className = 'vote-key';
      keyEl.textContent = String(i + 1);

      const contentEl = document.createElement('div');
      contentEl.className = 'vote-option-content';

      const nameEl = document.createElement('div');
      nameEl.className = 'vote-option-name';
      nameEl.textContent = opt.name;

      const trackEl = document.createElement('div');
      trackEl.className = 'vote-bar-track';
      const fillEl = document.createElement('div');
      fillEl.className = 'vote-bar-fill';
      trackEl.appendChild(fillEl);

      contentEl.appendChild(nameEl);
      contentEl.appendChild(trackEl);
      optEl.appendChild(keyEl);
      optEl.appendChild(contentEl);
      voteOptions.appendChild(optEl);

      voteState.elements.push({ nameEl: nameEl, fillEl: fillEl });
    }
  }

  function updateVote(data) {
    if (voteState.pendingHide) {
      clearTimeout(voteState.pendingHide);
      voteState.pendingHide = null;
    }

    const options = (data && Array.isArray(data.options)) ? data.options : [];
    const timeLeft = (data && typeof data.timeLeft === 'number') ? data.timeLeft : 0;

    if (options.length === 0 || timeLeft <= 0) {
      hideVote();
      return;
    }

    // If first time visible, build the structure
    if (!voteState.visible) {
      buildVoteOptions(options);
      votePanel.classList.remove('hidden');
      votePanel.classList.remove('fading');
      voteState.visible = true;
    } else {
      // If option count differs, rebuild
      if (voteState.elements.length !== Math.min(3, options.length)) {
        buildVoteOptions(options);
      } else {
        // Update names in case
        for (let i = 0; i < voteState.elements.length; i++) {
          const opt = options[i];
          if (opt && opt.name && voteState.elements[i]) {
            voteState.elements[i].nameEl.textContent = opt.name;
          }
        }
      }
    }

    // Update vote bars
    const totalVotes = options.reduce(function (sum, o) {
      return sum + Math.max(0, Number(o.votes) || 0);
    }, 0);

    for (let i = 0; i < voteState.elements.length; i++) {
      const opt = options[i];
      const votes = Math.max(0, Number(opt && opt.votes) || 0);
      const pct = totalVotes > 0 ? (votes / totalVotes) * 100 : 0;
      const fillEl = voteState.elements[i].fillEl;
      // Reset to 0 then animate so transitions trigger on first show
      if (fillEl.style.width === '') {
        fillEl.style.width = '0%';
        requestAnimationFrame(function (el, p) {
          return function () { el.style.width = p + '%'; };
        }(fillEl, pct));
      } else {
        fillEl.style.width = pct + '%';
      }
    }

    voteTime.textContent = Math.ceil(timeLeft) + 's';
  }

  function hideVote() {
    if (!voteState.visible) return;
    votePanel.classList.add('fading');
    voteState.pendingHide = setTimeout(function () {
      votePanel.classList.add('hidden');
      votePanel.classList.remove('fading');
      voteState.visible = false;
      voteState.elements = [];
      voteState.pendingHide = null;
    }, 320);
  }

  // ============== HITS ==============
  const hitGlowColors = {
    'REVERSED!':  '#ff00ff',
    'CRITICAL!':  '#ff4444',
    'HEADSHOT!':  '#ffaa00',
    'COMBO!':     '#00f0ff',
    'BONUS!':     '#00ff88',
    'PERFECT!':   '#ffcc00',
    'LUCKY!':     '#b44dff',
    'CHAOS!':     '#ff8c00'
  };

  function showHit(variant) {
    if (!variant || typeof variant !== 'string') return;
    const hit = document.createElement('div');
    hit.className = 'hit-notification';
    hit.textContent = variant;
    const glow = hitGlowColors[variant] || '#ffffff';
    hit.style.textShadow = '0 0 20px ' + glow + ', 0 0 40px ' + glow + ', 0 0 60px ' + glow;
    hitContainer.appendChild(hit);
    setTimeout(function () {
      if (hit.parentNode) hit.parentNode.removeChild(hit);
    }, 1550);
  }

  // ============== COUGARS ==============
  function updateCougars(count) {
    const c = Math.max(0, Math.floor(Number(count) || 0));
    if (c <= 0) {
      cougarCounter.classList.add('hidden');
      lastCougarCount = 0;
      return;
    }
    cougarCounter.classList.remove('hidden');
    if (c !== lastCougarCount) {
      cougarCount.textContent = String(c);
      cougarCounter.classList.remove('pulse');
      // Force reflow to restart animation
      void cougarCounter.offsetWidth;
      cougarCounter.classList.add('pulse');
      lastCougarCount = c;
    }
  }

  // ============== PHASE ==============
  function updatePhase(phase) {
    if (phase === currentPhase) return;

    if (phase === 'RUNNING' || phase == null) {
      phaseOverlay.classList.add('hidden');
      phaseOverlay.classList.remove('lobby', 'won', 'lost');
      phaseText.textContent = '';
      currentPhase = phase;
      return;
    }

    phaseOverlay.classList.remove('hidden', 'lobby', 'won', 'lost');
    if (phase === 'LOBBY') {
      phaseOverlay.classList.add('lobby');
      phaseText.textContent = 'WAITING FOR PLAYERS...';
    } else if (phase === 'WON') {
      phaseOverlay.classList.add('won');
      phaseText.textContent = 'VICTORY';
    } else if (phase === 'LOST') {
      phaseOverlay.classList.add('lost');
      phaseText.textContent = 'WASTED';
    } else {
      phaseOverlay.classList.add('hidden');
    }
    currentPhase = phase;
  }

  // ============== MESSAGE HANDLER ==============
  window.addEventListener('message', function (event) {
    const data = event.data;
    if (!data || typeof data !== 'object') return;

    switch (data.type) {
      case 'timer':
        updateTimer(data.progress);
        break;
      case 'effects':
        updateEffects(data.list);
        break;
      case 'vote':
        updateVote(data);
        break;
      case 'hit':
        showHit(data.variant);
        break;
      case 'cougars':
        updateCougars(data.count);
        break;
      case 'phase':
        updatePhase(data.phase);
        break;
      default:
        break;
    }
  });

})();
